# Layer 5 — Outcome Tracker

5.1 Outcome Record Schema
OUTCOME_RECORD {
  outcome_id:     uuid,
  ticker:         string,
  catalyst_id:    uuid,
  operating_mode: "LIVE|SHADOW|HISTORICAL",

  prediction_at_entry: {
    entry_date:               ISO8601,
    entry_price:              float,
    CCS_at_entry:             float,
    bull_probability:         float,
    base_probability:         float,
    bear_probability:         float,
    expected_value_pct:       float,
    bear_scenario_impact_pct: float,
    magnitude_class:          string,
    time_horizon:             string,
    sleeve_assignment:        string,
    recommended_position_$:   float,
    regime_at_entry:          string,
    institution_tier_signals: [int],
    TWOS_at_entry:            float,

    thesis_statement_at_entry:
      "350-token thesis statement as generated by Layer 4
       at time of recommendation — preserved verbatim
       for postmortem analysis and calibration review.
       Allows comparison of stated thesis against
       actual outcome to identify systematic reasoning
       errors in the LLM extraction pipeline."
  },

  execution: {
    actual_position_$:    float,
    slippage_applied_pct: float,
    entry_method:         "market|limit"
  },

  resolution: {
    catalyst_resolved:          boolean,
    resolution_date:            ISO8601,
    actual_scenario:            "bull|base|bear|unresolved",
    actual_price_T_resolution:  float,
    actual_price_impact_pct:    float
  },

  exit: {
    exit_date:    ISO8601,
    exit_price:   float,
    exit_trigger: "score_collapse|dependency_violated|
                   valuation_realisation|survival_risk|
                   institutional_exit|TWOS_deterioration|
                   fixed_horizon",
    CCS_at_exit:  float,
    holding_days: int
  },

  measurement_outputs: {

    brier_score: float,
    // Output A — catalyst prediction accuracy
    // BS = (bull_p - bull_occurred)²
    //    + (base_p - base_occurred)²
    //    + (bear_p - bear_occurred)²

    price_impact_MAE: float,
    // Output B — price impact accuracy
    // abs(actual_price_impact - predicted_scenario_impact)

    round_trip_return_pct:    float,
    // Output C — T_entry to T_exit
    fixed_horizon_return_pct: float,
    // Output C — T_entry to T_entry + horizon_midpoint
    exit_timing_delta_pct:    float,
    // positive: exit timing added value
    // negative: exit timing destroyed value

    sharpe_contribution: float,
    // (round_trip_return - risk_free_rate)
    //   / position_return_volatility — annualised

    regime_at_resolution: string
  }
}

5.2 Measurement Clocks
MEASUREMENT_CLOCKS {

  T_entry: {
    definition: "day of Layer 4 recommendation generation",
    price:      "closing price",
    slippage: {
      ADV_above_1M:   "0%",
      ADV_500k_to_1M: "0.5%",
      ADV_below_500k: "1.0%"
    }
  },

  T_resolution: {
    definition:
      "closing price on day catalyst resolves:
       FDA decision date, Phase 3 readout date,
       M&A announcement date, earnings date",
    for_undated_catalysts:
      "T_thesis_change = date when Layer 3 registry
       records probability update >10pp in any direction",
    primary_measurement_for: "Output A and Output B"
  },

  T_exit: {
    definition:
      "closing price on day first sell trigger fires,
       OR T_entry + fixed_horizon if no trigger fires",
    fixed_horizon_values: {
      near_term:   "T_entry + 75 days",
      medium_term: "T_entry + 195 days",
      structural:  "T_entry + 405 days"
    },
    primary_measurement_for: "Output C"
  },

  dual_measurement_rule:
    "always record BOTH T_exit return AND fixed_horizon return
     exit_timing_delta reveals whether sell triggers add value
     if exit_timing_delta is systematically negative:
       sell triggers are destroying value — investigate"
}

5.3 Survivorship Bias Prevention
SURVIVORSHIP_BIAS_RULES {

  price_data_source:
    "Polygon.io or Nasdaq Data Link
     with delisted ticker history included
     survivorship-bias-free dataset required"

  delisted_ticker_handling:
    "if ticker delisted during holding period:
     record exit_price as last available trading price
     or $0.01 if trading halted with no recovery
     record exit_trigger as dependency_violated
     do not exclude from outcome calculations"

  bankruptcy_handling:
    "record as bear scenario resolution
     price_impact = -100%
     include in Brier score calculation
     critical for pre-revenue biotech calibration"
}
